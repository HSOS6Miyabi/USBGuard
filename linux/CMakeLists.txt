add_library(usbguard_linux_core STATIC core.cpp core.hpp)
target_compile_features(usbguard_linux_core PUBLIC cxx_std_17)
target_include_directories(usbguard_linux_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(usbguard_linux_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(USBGuardLinuxCLI cli.cpp)
target_link_libraries(USBGuardLinuxCLI PRIVATE usbguard_linux_core)
target_compile_options(USBGuardLinuxCLI PRIVATE -Wall -Wextra -Wpedantic)

option(USBGUARD_LINUX_GUI "Build GTK 3 window supporting Wayland and X11" ON)
if(USBGUARD_LINUX_GUI)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(GTK3 IMPORTED_TARGET gtk+-3.0)
    endif()
    if(GTK3_FOUND)
        add_executable(USBGuardLinux gui.cpp)
        target_compile_features(USBGuardLinux PRIVATE cxx_std_17)
        target_link_libraries(USBGuardLinux PRIVATE usbguard_linux_core PkgConfig::GTK3)
        target_compile_options(USBGuardLinux PRIVATE -Wall -Wextra -Wpedantic)
        if(BUILD_TESTING)
            find_program(XVFB_RUN xvfb-run)
            if(XVFB_RUN)
                add_test(NAME linux_gui_smoke COMMAND "${XVFB_RUN}" -a
                         "$<TARGET_FILE:USBGuardLinux>" --ui-self-test)
                set_tests_properties(linux_gui_smoke PROPERTIES TIMEOUT 20)
            endif()
        endif()
    else()
        message(STATUS "GTK 3 development libraries unavailable: building Linux CLI only")
    endif()
endif()

if(BUILD_TESTING)
    add_executable(linux_core_tests tests.cpp)
    target_link_libraries(linux_core_tests PRIVATE usbguard_linux_core)
    add_test(NAME linux_core_tests COMMAND linux_core_tests)
    add_test(NAME linux_cli_help COMMAND "$<TARGET_FILE:USBGuardLinuxCLI>" --help)
    add_test(NAME linux_cli_reject_unsafe_device COMMAND "$<TARGET_FILE:USBGuardLinuxCLI>" --device /dev/../sda --protect)
    set_tests_properties(linux_cli_reject_unsafe_device PROPERTIES WILL_FAIL TRUE)
endif()
